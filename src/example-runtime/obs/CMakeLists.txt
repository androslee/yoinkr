cmake_minimum_required(VERSION 3.22)
project(yoinkr_annotation VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 11)
set(CMAKE_AUTOMOC ON)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_library(yoinkr-annotation MODULE annotation.cpp smoothing.c)
target_link_libraries(yoinkr-annotation PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets)
if(UNIX)
 target_link_libraries(yoinkr-annotation PRIVATE m)
endif()
set_target_properties(yoinkr-annotation PROPERTIES PREFIX "")
install(TARGETS yoinkr-annotation LIBRARY DESTINATION lib/obs-plugins)
